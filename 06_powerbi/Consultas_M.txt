En Power Query, crea una consulta en blanco por cada bloque, abre Editor avanzado y pega el bloque correspondiente. Reemplaza SERVIDOR_SQL. Autentícate con el cuadro de credenciales de Power BI; no pongas una contraseña aquí.

Consulta: FactEntregas
let
    Fuente = Sql.Database("SERVIDOR_SQL", "BancoAlimentos_CNPV"),
    Vista = Fuente{[Schema="CNPV", Item="vw_DetalleEntregas"]}[Data],
    Tipos = Table.TransformColumnTypes(Vista, {{"FechaEntrega", type date}, {"CantidadEntregada", Int64.Type}})
in
    Tipos

Consulta: Inventario
let
    Fuente = Sql.Database("SERVIDOR_SQL", "BancoAlimentos_CNPV"),
    Alimentos = Fuente{[Schema="CNPV", Item="Alimentos"]}[Data],
    Categorias = Fuente{[Schema="CNPV", Item="CategoriaAlimento"]}[Data],
    Cruce = Table.NestedJoin(Alimentos, {"IdCategoria"}, Categorias, {"IdCategoria"}, "DatosCategoria", JoinKind.LeftOuter),
    Expandido = Table.ExpandTableColumn(Cruce, "DatosCategoria", {"NombreCategoria"}, {"Categoria"}),
    Columnas = Table.SelectColumns(Expandido, {"IdAlimento", "Nombre", "Categoria", "Cantidad", "UnidadMedida", "FechaVencimiento"}),
    Tipos = Table.TransformColumnTypes(Columnas, {{"Cantidad", Int64.Type}, {"FechaVencimiento", type date}})
in
    Tipos
